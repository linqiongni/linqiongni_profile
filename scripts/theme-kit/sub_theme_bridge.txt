<script>(function(){
  if(window.__themeBridge)return;window.__themeBridge=1;
  var r=document.documentElement;
  function apply(m){
    m=(m==='light')?'light':'dark';
    r.setAttribute('data-theme',m);
    try{ r.setAttribute('data-theme-default',m); }catch(e){}
    try{ window.dispatchEvent(new CustomEvent('site-theme',{detail:m})); }catch(e){}
  }
  // 主站导航栏的太阳/月亮按钮 → ThemeIframe 用 postMessage 把模式推进来
  window.addEventListener('message',function(e){
    var d=e.data||{};
    if(d.type==='theme'&&(d.mode==='light'||d.mode==='dark')){ apply(d.mode); }
  });
  function standalone(){ try{ return window.parent===window; }catch(e){ return false; } }
  function addBtn(){
    if(!standalone()||document.getElementById('theme-toggle-btn'))return;
    // 子站自带的 assets/app.js 已经提供了主题按钮（#themeBtn）时不要再加，避免出现两个。
    if(document.getElementById('themeBtn'))return;
    var b=document.createElement('button');
    b.id='theme-toggle-btn';b.type='button';
    b.style.cssText=[
      'position:fixed','top:56px','right:16px','z-index:9999',
      'padding:6px 14px','border-radius:999px','cursor:pointer',
      'font-size:13px','font-family:inherit',
      'border:1px solid var(--line,#CFCABE)',
      'background:var(--card,#FFFFFF)',
      'color:var(--txt,#23201B)',
      'box-shadow:0 1px 3px rgba(0,0,0,.08)',
      'transition:border-color .15s,color .15s'
    ].join(';');
    if(window.matchMedia&&window.matchMedia('(max-width: 640px)').matches){
      b.style.top='auto';b.style.bottom='16px';b.style.right='12px';b.style.padding='5px 10px';
    }
    function paint(){ b.textContent = r.getAttribute('data-theme')==='dark' ? '切换浅色' : '切换深色'; }
    b.addEventListener('click',function(){
      apply(r.getAttribute('data-theme')==='dark'?'light':'dark');
      paint();
    });
    paint();
    (document.body||r).appendChild(b);
  }
  if(document.readyState==='loading'){ document.addEventListener('DOMContentLoaded',addBtn); }
  else { addBtn(); }
})();</script>
